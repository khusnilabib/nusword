   239→function ChapterTree({
   240→  bookId,
   241→  chapters,
   242→  activeChapterId,
   243→  onSelectChapter,
   244→}: {
   245→  bookId: string;
   246→  chapters: ChapterNode[];
   247→  activeChapterId: string | null;
   248→  onSelectChapter: (id: string) => void;
   249→}) {
   250→  const createChapter = useCreateChapter(bookId);
   251→
   252→  const handleAddChapter = () => {
   253→    createChapter.mutate(
   254→      { title: `Chapter ${chapters.length + 1}` },
   255→      {
   256→        onSuccess: (chapter) => onSelectChapter(chapter.id),
   257→        onError: () => toast.error("Failed to create chapter"),
   258→      },
   259→    );
   260→  };
   261→
   262→  return (
   263→    <div className="space-y-2">
   264→      <button
   265→        type="button"
   266→        onClick={handleAddChapter}
   267→        disabled={createChapter.isPending}
   268→        className="flex w-full cursor-pointer items-center justify-center gap-2 rounded border border-dashed border-outline-variant py-2 text-body-ui-md text-primary transition-colors hover:bg-surface-container-low disabled:opacity-50"
   269→      >
   270→        <Icon name={createChapter.isPending ? "progress_activity" : "add"} size={16} />
   271→        Add Chapter
   272→      </button>
   273→
   274→      {chapters.length === 0 ? (
   275→        <div className="flex flex-col items-center gap-2 py-8 text-center">
   276→          <Icon name="menu_book" size={28} className="text-outline-variant" />
   277→          <p className="text-body-ui-md text-on-surface-variant">No chapters yet</p>
   278→          <p className="text-label-ui-sm text-outline">
   279→            Add your first chapter to start building the book.
   280→          </p>
   281→        </div>
   282→      ) : (
   283→        <div className="space-y-0.5">
   284→          {chapters.map((chapter, idx) => (
   285→            <ChapterTreeItem
   286→              key={chapter.id}
   287→              bookId={bookId}
   288→              chapter={chapter}
   289→              index={idx}
   290→              active={activeChapterId === chapter.id}
   291→              onSelect={onSelectChapter}
   292→            />
   293→          ))}
   294→        </div>
   295→      )}
   296→    </div>
   297→  );
   298→}

   300→function ChapterTreeItem({
   301→  bookId,
   302→  chapter,
   303→  index,
   304→  active,
   305→  onSelect,
   306→}: {
   307→  bookId: string;
   308→  chapter: ChapterNode;
   309→  index: number;
   310→  active: boolean;
   311→  onSelect: (id: string) => void;
   312→}) {
   313→  const deleteChapter = useDeleteChapter(bookId);
   314→  const [expanded, setExpanded] = React.useState(true);
   315→
   316→  return (
   317→    <div>
   318→      <div
   319→        className={cn(
   320→          "group flex items-center gap-1 rounded py-1.5 pr-2 transition-colors cursor-pointer",
   321→          active ? "bg-surface-container-high text-primary" : "text-on-surface-variant hover:bg-surface-container-low",
   322→        )}
   323→        onClick={() => onSelect(chapter.id)}
   324→      >
   325→        <span className="text-mono-ui w-6 shrink-0 text-center text-outline">
   326→          {index + 1}
   327→        </span>
   328→        <Icon name="description" size={16} className="shrink-0" />
   329→        <span className="text-body-ui-md flex-1 truncate">{chapter.title}</span>
   330→        {chapter.children.length > 0 && (
   331→          <button
   332→            type="button"
   333→            onClick={(e) => {
   334→              e.stopPropagation();
   335→              setExpanded((v) => !v);
   336→            }}
   337→            className="flex size-5 items-center justify-center rounded hover:bg-surface-container"
   338→          >
   339→            <Icon name={expanded ? "expand_more" : "chevron_right"} size={14} />
   340→          </button>
   341→        )}
   342→        <button
   343→          type="button"
   344→          onClick={(e) => {
   345→            e.stopPropagation();
   346→            if (confirm(`Delete chapter "${chapter.title}"?`)) {
   347→              deleteChapter.mutate(chapter.id, {
   348→                onSuccess: () => toast.success("Chapter deleted"),
   349→                onError: () => toast.error("Failed to delete chapter"),
   350→              });
   351→            }
   352→          }}
   353→          className="flex size-5 items-center justify-center rounded text-on-surface-variant opacity-0 hover:bg-error-container/30 hover:text-error group-hover:opacity-100"
   354→          aria-label="Delete chapter"
   355→        >
   356→          <Icon name="delete" size={14} />
   357→        </button>
   358→      </div>
   359→
   360→      {expanded && chapter.children.length > 0 && (
   361→        <div className="ml-6 border-l border-outline-variant pl-2">
   362→          {chapter.children.map((child, idx) => (
   363→            <ChapterTreeItem
   364→              key={child.id}
   365→              bookId={bookId}
   366→              chapter={child}
   367→              index={idx}
   368→              active={active}
   369→              onSelect={onSelect}
   370→            />
   371→          ))}
   372→        </div>
   373→      )}
   374→    </div>
   375→  );
   376→}

   378→/* ================================================================
   379→   Chapter Editor — loads the chapter's document into Tiptap
   380→   ================================================================ */
   381→
   382→function ChapterEditor({
   383→  bookId,
   384→  chapterId,
   385→  chapters,
   386→}: {
   387→  bookId: string;
   388→  chapterId: string;
   389→  chapters: ChapterNode[];
   390→}) {
   391→  // Find the chapter and its document.
   392→  const findChapter = (nodes: ChapterNode[]): ChapterNode | null => {
   393→    for (const n of nodes) {
   394→      if (n.id === chapterId) return n;
   395→      const found = findChapter(n.children);
   396→      if (found) return found;
   397→    }
   398→    return null;
   399→  };
   400→  const chapter = findChapter(chapters);
   401→  const documentId = chapter?.documentId;
   402→
   403→  const { data: doc, isLoading } = useDocument(documentId);
   404→  const updateChapter = useUpdateChapter(bookId);
   405→
   406→  const [title, setTitle] = React.useState(chapter?.title ?? "");
   407→  const [content, setContent] = React.useState<JSONContent | null>(null);
   408→  const [hydrated, setHydrated] = React.useState(false);
   409→  const editorRef = React.useRef<NuswordEditorHandle | null>(null);
   410→
   411→  React.useEffect(() => {
   412→    if (doc) {
   413→      setContent(doc.content);
   414→      setHydrated(true);
   415→    } else {
   416→      setHydrated(false);
   417→    }
   418→  }, [doc]);
   419→
   420→  React.useEffect(() => {
   421→    setTitle(chapter?.title ?? "");
   422→  }, [chapter?.title]);
   423→
   424→  const updateMutation = useUpdateDocument(documentId ?? "");
   425→  const { saveState } = useAutosave({
   426→    documentId,
   427→    title,
   428→    content,
   429→    settings: doc?.settings ?? null,
   430→    ready: hydrated,
   431→  });
   432→
   433→  if (!chapter) {
   434→    return (
   435→      <div className="flex flex-1 items-center justify-center">
   436→        <p className="text-body-ui-md text-on-surface-variant">Chapter not found</p>
   437→      </div>
   438→    );
   439→  }
   440→
   441→  if (isLoading || !doc || !content) {
   442→    return (
   443→      <div className="flex flex-1 items-center justify-center">
   444→        <Icon name="progress_activity" size={24} className="animate-spin text-primary" />
   445→      </div>
   446→    );
   447→  }
   448→
   449→  const saveLabel =
   450→    saveState === "saving" ? "Saving…" : saveState === "saved" ? "Saved" : saveState === "error" ? "Save failed" : "Saved";
   451→
   452→  return (
   453→    <main className="relative flex flex-1 flex-col items-center overflow-auto bg-surface-container-low px-4 py-10">
   454→      {/* Chapter header */}
   455→      <div className="mb-4 flex w-[210mm] max-w-full items-center justify-between">
   456→        <div className="flex items-center gap-2">
   457→          <span className="text-label-ui-sm uppercase tracking-wider text-on-surface-variant">
   458→            Chapter
   459→          </span>
   460→          <input
   461→            type="text"
   462→            value={title}
   463→            onChange={(e) => setTitle(e.target.value)}
   464→            onBlur={() => {
   465→              if (title !== chapter.title) {
   466→                updateChapter.mutate({ chapterId, title });
   467→              }
   468→            }}
   469→            className="text-body-ui-md border-b border-transparent bg-transparent font-semibold text-on-surface focus:border-primary focus:outline-none"
   470→          />
   471→        </div>
   472→        <span className={cn("text-label-ui-sm flex items-center gap-1", saveState === "error" ? "text-error" : "text-outline")}>
   473→          <Icon name={saveState === "saving" ? "progress_activity" : "cloud_done"} size={14} />
   474→          {saveLabel}
   475→        </span>
   476→      </div>
   477→
   478→      {/* Paper + editor */}
   479→      <article
   480→        className="paper-shadow relative z-0 mx-auto mb-20 flex w-[210mm] max-w-full min-h-[297mm] origin-top flex-col bg-white p-[25.4mm]"
   481→      >
   482→        <NuswordEditor
   483→          key={documentId}
   484→          initialContent={content}
   485→          onChange={setContent}
   486→          editorRef={editorRef}
   487→          fontSizePt={doc.settings.fontSizePt}
   488→          lineHeight={doc.settings.lineHeight}
   489→          dir={doc.settings.languageDirection}
   490→        />
   491→      </article>
   492→    </main>
   493→  );
   494→}

   515→
   516→function MatterPanel({ bookId, matter }: { bookId: string; matter: "front" | "back" }) {
   517→  const { data: book } = useBook(bookId);
   518→  const updateBook = useUpdateBook(bookId);
   519→  const entries = matter === "front" ? book?.frontMatter ?? [] : book?.backMatter ?? [];
   520→  const types = matter === "front" ? FRONT_MATTER_TYPES : BACK_MATTER_TYPES;
   521→
   522→  if (!book) return null;
   523→
   524→  const handleToggle = (entryId: string) => {
   525→    const updated = entries.map((e) =>
   526→      e.id === entryId ? { ...e, enabled: !e.enabled } : e,
   527→    );
   528→    updateBook.mutate(
   529→      matter === "front" ? { frontMatter: updated } : { backMatter: updated },
   530→    );
   531→  };
   532→
   533→  const handleAdd = (type: FrontMatterType | BackMatterType, label: string) => {
   534→    const newEntry: BookMatterEntry = {
   535→      id: crypto.randomUUID(),
   536→      type,
   537→      title: label,
   538→      enabled: true,
   539→    };
   540→    const updated = [...entries, newEntry];
   541→    updateBook.mutate(
   542→      matter === "front" ? { frontMatter: updated } : { backMatter: updated },
   543→    );
   544→  };
   545→
   546→  return (
   547→    <div className="space-y-3">
   548→      <div className="text-label-ui-sm text-on-surface-variant">
   549→        {matter === "front" ? "Front Matter" : "Back Matter"}
   550→      </div>
   551→
   552→      {/* Existing entries */}
   553→      {entries.length > 0 && (
   554→        <div className="space-y-1">
   555→          {entries.map((entry) => (
   556→            <div
   557→              key={entry.id}
   558→              className={cn(
   559→                "flex items-center gap-2 rounded border px-2 py-1.5",
   560→                entry.enabled
   561→                  ? "border-outline-variant bg-surface-container-lowest"
   562→                  : "border-outline-variant/50 bg-surface-container-low/50 opacity-60",
   563→              )}
   564→            >
   565→              <Icon
   566→                name={types.find((t) => t.type === entry.type)?.icon ?? "article"}
   567→                size={16}
   568→                className="text-on-surface-variant"
   569→              />
   570→              <span className="text-body-ui-md flex-1 truncate text-on-surface">
   571→                {entry.title}
   572→              </span>
   573→              <button
   574→                type="button"
   575→                onClick={() => handleToggle(entry.id)}
   576→                className={cn(
   577→                  "flex size-6 cursor-pointer items-center justify-center rounded",
   578→                  entry.enabled ? "text-primary" : "text-on-surface-variant",
   579→                )}
   580→              >
   581→                <Icon name={entry.enabled ? "toggle_on" : "toggle_off"} size={18} />
   582→              </button>
   583→            </div>
   584→          ))}
   585→        </div>
