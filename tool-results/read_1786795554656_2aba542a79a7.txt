   184→function OrganizationCard({
   185→  org,
   186→  onManage,
   187→}: {
   188→  org: NuswordOrganization;
   189→  onManage: () => void;
   190→}) {
   191→  const deleteMutation = useDeleteOrganization();
   192→  const [confirming, setConfirming] = React.useState(false);
   193→  const roleMeta = ROLE_META[org.myRole];
   194→  const canDelete = org.myRole === "owner";
   195→
   196→  const handleDelete = () => {
   197→    deleteMutation.mutate(org.id, {
   198→      onSuccess: () => {
   199→        toast.success(`"${org.name}" deleted`);
   200→        setConfirming(false);
   201→      },
   202→      onError: (err) => {
   203→        toast.error(err instanceof Error ? err.message : "Failed to delete organization");
   204→      },
   205→    });
   206→  };
   207→
   208→  return (
   209→    <article className="group flex flex-col rounded-lg border border-outline-variant bg-surface p-5 transition-all hover:border-outline">
   210→      {/* Title row */}
   211→      <div className="flex items-start justify-between gap-3">
   212→        <div className="flex min-w-0 items-center gap-3">
   213→          <div className="flex size-10 shrink-0 items-center justify-center rounded bg-primary-container text-on-primary-container">
   214→            <Icon name={roleMeta.icon} size={22} />
   215→          </div>
   216→          <div className="min-w-0">
   217→            <h3 className="text-headline-ui-md truncate text-on-surface">
   218→              {org.name}
   219→            </h3>
   220→            <span className="text-mono-ui text-outline">
   221→              {org.slug}
   222→            </span>
   223→          </div>
   224→        </div>
   225→        <RoleBadge role={org.myRole} />
   226→      </div>
   227→
   228→      {/* Description */}
   229→      <p className="text-body-ui-md mt-3 line-clamp-2 min-h-[2.5rem] text-on-surface-variant">
   230→        {org.description || "No description provided."}
   231→      </p>
   232→
   233→      {/* Stats */}
   234→      <div className="mt-4 grid grid-cols-3 gap-2">
   235→        <StatChip icon="group" label="Members" value={org.memberCount} />
   236→        <StatChip icon="description" label="Docs" value={org.documentCount} />
   237→        <StatChip icon="menu_book" label="Books" value={org.bookCount} />
   238→      </div>
   239→
   240→      {/* Footer */}
   241→      <div className="mt-5 flex items-center justify-between gap-2 border-t border-outline-variant/50 pt-4">
   242→        <span className="text-label-ui-sm text-outline">
   243→          Created {relativeTime(org.createdAt)}
   244→        </span>
   245→        <div className="flex items-center gap-2">
   246→          {canDelete && !confirming ? (
   247→            <button
   248→              type="button"
   249→              aria-label={`Delete ${org.name}`}
   250→              onClick={() => setConfirming(true)}
   251→              className="flex size-9 cursor-pointer items-center justify-center rounded text-on-surface-variant transition-colors hover:bg-error-container/40 hover:text-on-error-container"
   252→            >
   253→              <Icon name="delete" size={18} />
   254→            </button>
   255→          ) : null}
   256→          <button
   257→            type="button"
   258→            onClick={onManage}
   259→            className="inline-flex h-9 cursor-pointer items-center gap-2 rounded border border-outline-variant px-3 text-body-ui-md font-medium text-on-surface transition-colors hover:bg-surface-container-low"
   260→          >
   261→            <Icon name="manage_accounts" size={16} />
   262→            Manage
   263→          </button>
   264→        </div>
   265→      </div>
   266→
   267→      {/* Confirm-delete inline panel */}
   268→      {confirming && (
   269→        <div className="mt-3 rounded border border-error-container/60 bg-error-container/20 p-3">
   270→          <p className="text-body-ui-md text-on-surface">
   271→            Delete <span className="font-semibold">{org.name}</span>? This
   272→            cannot be undone.
   273→          </p>
   274→          <div className="mt-3 flex justify-end gap-2">
   275→            <button
   276→              type="button"
   277→              onClick={() => setConfirming(false)}
   278→              disabled={deleteMutation.isPending}
   279→              className="cursor-pointer rounded px-3 py-1.5 text-body-ui-md text-on-surface-variant transition-colors hover:bg-surface-container-low disabled:opacity-50"
   280→            >
   281→              Cancel
   282→            </button>
   283→            <button
   284→              type="button"
   285→              onClick={handleDelete}
   286→              disabled={deleteMutation.isPending}
   287→              className="inline-flex h-8 cursor-pointer items-center gap-1.5 rounded bg-destructive px-3 text-body-ui-md font-medium text-white transition-colors hover:bg-destructive/90 disabled:cursor-wait disabled:opacity-50"
   288→            >
   289→              <Icon
   290→                name={deleteMutation.isPending ? "progress_activity" : "delete"}
   291→                size={16}
   292→                className={deleteMutation.isPending ? "animate-spin" : ""}
   293→              />
   294→              Delete
   295→            </button>
   296→          </div>
   297→        </div>
   298→      )}
   299→    </article>
   300→  );
   301→}

   356→
   357→function CreateOrganizationDialog({
   358→  open,
   359→  onOpenChange,
   360→}: {
   361→  open: boolean;
   362→  onOpenChange: (open: boolean) => void;
   363→}) {
   364→  const createMutation = useCreateOrganization();
   365→  const [name, setName] = React.useState("");
   366→  const [description, setDescription] = React.useState("");
   367→
   368→  // Reset on close.
   369→  React.useEffect(() => {
   370→    if (!open) {
   371→      // Defer reset so the closing animation doesn't show empty fields.
   372→      const t = setTimeout(() => {
   373→        setName("");
   374→        setDescription("");
   375→      }, 150);
   376→      return () => clearTimeout(t);
   377→    }
   378→  }, [open]);
   379→
   380→  const nameTrim = name.trim();
   381→  const canSubmit = nameTrim.length > 0 && !createMutation.isPending;
   382→
   383→  const handleSubmit = (e: React.FormEvent) => {
   384→    e.preventDefault();
   385→    if (!canSubmit) return;
   386→    createMutation.mutate(
   387→      {
   388→        name: nameTrim,
   389→        description: description.trim() || undefined,
   390→      },
   391→      {
   392→        onSuccess: (org) => {
   393→          toast.success(`Organization "${org.name}" created`);
   394→          onOpenChange(false);
   395→        },
   396→        onError: (err) => {
   397→          toast.error(
   398→            err instanceof Error
   399→              ? err.message
   400→              : "Failed to create organization",
   401→          );
   402→        },
   403→      },
   404→    );
   405→  };
   406→
   407→  return (
   408→    <Dialog open={open} onOpenChange={onOpenChange}>
   409→      <DialogContent className="max-w-md bg-surface p-0">
   410→        <DialogHeader className="border-b border-outline-variant p-4">
   411→          <DialogTitle className="text-headline-ui-md text-on-surface">
   412→            Create Organization
   413→          </DialogTitle>
   414→          <DialogDescription className="text-body-ui-md text-on-surface-variant">
   415→            Organizations group members, documents, and books for a team or
   416→            project.
   417→          </DialogDescription>
   418→        </DialogHeader>
   419→
   420→        <form onSubmit={handleSubmit} className="space-y-4 p-4">
   421→          <div className="space-y-1.5">
   422→            <label
   423→              htmlFor="org-name"
   424→              className="text-label-ui-sm block uppercase tracking-wider text-on-surface-variant"
   425→            >
   426→              Name <span className="text-destructive">*</span>
   427→            </label>
   428→            <input
   429→              id="org-name"
   430→              type="text"
   431→              required
   432→              autoFocus
   433→              maxLength={100}
   434→              value={name}
   435→              onChange={(e) => setName(e.target.value)}
   436→              placeholder="e.g. Andalus Press"
   437→              className="h-10 w-full rounded border border-outline-variant bg-surface-container-lowest px-3 text-body-ui-md text-on-surface transition-colors placeholder:text-on-surface-variant/60 focus:border-primary focus:outline-none"
   438→            />
   439→          </div>
   440→
   441→          <div className="space-y-1.5">
   442→            <label
   443→              htmlFor="org-description"
   444→              className="text-label-ui-sm block uppercase tracking-wider text-on-surface-variant"
   445→            >
   446→              Description
   447→            </label>
   448→            <textarea
   449→              id="org-description"
   450→              rows={3}
   451→              maxLength={500}
   452→              value={description}
   453→              onChange={(e) => setDescription(e.target.value)}
   454→              placeholder="What is this organization for?"
   455→              className="w-full resize-none rounded border border-outline-variant bg-surface-container-lowest px-3 py-2 text-body-ui-md text-on-surface transition-colors placeholder:text-on-surface-variant/60 focus:border-primary focus:outline-none"
   456→            />
   457→          </div>
   458→        </form>
   459→
   460→        <DialogFooter className="border-t border-outline-variant p-4 sm:justify-end">
   461→          <button
   462→            type="button"
   463→            onClick={() => onOpenChange(false)}
   464→            disabled={createMutation.isPending}
   465→            className="cursor-pointer rounded border border-outline-variant px-4 py-1.5 text-body-ui-md text-on-surface-variant transition-colors hover:bg-surface-container-low disabled:opacity-50"
   466→          >
   467→            Cancel
   468→          </button>
   469→          <button
   470→            type="button"
   471→            onClick={handleSubmit}
   472→            disabled={!canSubmit}
   473→            className="inline-flex h-9 cursor-pointer items-center gap-2 rounded bg-primary px-4 text-body-ui-md font-medium text-on-primary transition-colors hover:bg-primary-container disabled:cursor-wait disabled:opacity-50"
   474→          >
   475→            <Icon
   476→              name={createMutation.isPending ? "progress_activity" : "add"}
   477→              size={16}
   478→              className={createMutation.isPending ? "animate-spin" : ""}
   479→            />
   480→            {createMutation.isPending ? "Creating…" : "Create"}
   481→          </button>
   482→        </DialogFooter>
   483→      </DialogContent>
   484→    </Dialog>
   485→  );
   486→}

   488→/* ------------------------------------------------------------------ */
   489→/* Manage-members dialog                                               */
   490→/* ------------------------------------------------------------------ */
   491→
   492→function ManageMembersDialog({
   493→  org,
   494→  open,
   495→  onOpenChange,
   496→}: {
   497→  org: NuswordOrganization | null;
   498→  open: boolean;
   499→  onOpenChange: (open: boolean) => void;
   500→}) {
   501→  // Keep a local copy of the org so the dialog content stays mounted during
   502→  // the close transition (the parent clears `managedOrgId` immediately on
   503→  // `onOpenChange(false)`, which would otherwise blank the dialog before the
   504→  // exit animation finishes).
   505→  const [stagedOrg, setStagedOrg] = React.useState<NuswordOrganization | null>(
   506→    org,
   507→  );
   508→  React.useEffect(() => {
   509→    if (open && org) setStagedOrg(org);
   510→  }, [open, org]);
   511→
   512→  const activeOrg = stagedOrg ?? org;
   513→  // Hooks must be called unconditionally — pass null when there's no org so
   514→  // the inner query disables itself.
   515→  const orgId = activeOrg?.id ?? "";
   516→  const { data: members = [], isLoading } = useOrgMembers(
   517→    open ? orgId : null,
   518→  );
   519→
   520→  if (!activeOrg) {
   521→    // Render an invisible dialog host so transitions still work.
   522→    return (
   523→      <Dialog open={open} onOpenChange={onOpenChange}>
   524→        <DialogContent className="hidden" />
   525→      </Dialog>
   526→    );
   527→  }
   528→
   529→  const canManage =
   530→    activeOrg.myRole === "owner" || activeOrg.myRole === "admin";
   531→
   532→  return (
   533→    <Dialog open={open} onOpenChange={onOpenChange}>
   534→      <DialogContent className="max-w-2xl bg-surface p-0">
   535→        <DialogHeader className="border-b border-outline-variant p-4">
   536→          <DialogTitle className="text-headline-ui-md flex items-center gap-2 text-on-surface">
   537→            <Icon name="group" size={20} className="text-primary" />
   538→            {activeOrg.name}
   539→          </DialogTitle>
   540→          <DialogDescription className="text-body-ui-md text-on-surface-variant">
   541→            Invite teammates and manage their roles.
   542→          </DialogDescription>
   543→        </DialogHeader>
   544→
   545→        <div className="max-h-[70vh] space-y-6 overflow-y-auto p-4">
   546→          {/* Invite form */}
   547→          <InviteMemberForm orgId={orgId} canManage={canManage} />
   548→
   549→          {/* Members list */}
   550→          <div>
   551→            <div className="mb-2 flex items-center justify-between">
   552→              <h4 className="text-label-ui-sm uppercase tracking-wider text-on-surface-variant">
   553→                Members
   554→                {members.length > 0 && (
   555→                  <span className="ml-2 text-outline">{members.length}</span>
   556→                )}
   557→              </h4>
   558→            </div>
   559→
   560→            {isLoading ? (
   561→              <div className="space-y-2">
   562→                {[0, 1].map((i) => (
   563→                  <div
   564→                    key={i}
   565→                    className="h-14 animate-pulse rounded bg-surface-container-low"
   566→                  />
   567→                ))}
   568→              </div>
   569→            ) : members.length === 0 ? (
   570→              <p className="text-body-ui-md rounded border border-dashed border-outline-variant/50 px-4 py-6 text-center text-on-surface-variant">
   571→                No members found.
   572→              </p>
   573→            ) : (
   574→              <ul className="space-y-1.5">
   575→                {members.map((m) => (
   576→                  <MemberRow
   577→                    key={m.id}
   578→                    member={m}
   579→                    orgId={orgId}
   580→                    canManage={canManage}
   581→                  />
   582→                ))}
   583→              </ul>
   584→            )}
   585→          </div>
   586→        </div>
   587→
   588→        <DialogFooter className="border-t border-outline-variant p-4 sm:justify-end">
   589→          <button
   590→            type="button"
   591→            onClick={() => onOpenChange(false)}
   592→            className="cursor-pointer rounded border border-outline-variant px-4 py-1.5 text-body-ui-md text-on-surface-variant transition-colors hover:bg-surface-container-low"
   593→          >
   594→            Close
   595→          </button>
   596→        </DialogFooter>
   597→      </DialogContent>
   598→    </Dialog>
   599→  );
   600→}

   603→/* Invite form                                                         */
   604→/* ------------------------------------------------------------------ */
   605→
   606→function InviteMemberForm({
   607→  orgId,
   608→  canManage,
   609→}: {
   610→  orgId: string;
   611→  canManage: boolean;
   612→}) {
   613→  const inviteMutation = useInviteMember(orgId);
   614→  const [email, setEmail] = React.useState("");
   615→  const [name, setName] = React.useState("");
   616→  const [role, setRole] = React.useState<OrgRole>("editor");
   617→
   618→  const emailTrim = email.trim();
   619→  const canSubmit =
   620→    canManage &&
   621→    /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailTrim) &&
   622→    !inviteMutation.isPending;
   623→
   624→  const handleInvite = (e: React.FormEvent) => {
   625→    e.preventDefault();
   626→    if (!canSubmit) return;
   627→    inviteMutation.mutate(
   628→      {
   629→        email: emailTrim,
   630→        name: name.trim() || undefined,
   631→        role,
   632→      },
   633→      {
   634→        onSuccess: (m) => {
   635→          toast.success(`Invited ${m.email} as ${ROLE_META[m.role].label}`);
   636→          setEmail("");
   637→          setName("");
   638→          setRole("editor");
   639→        },
   640→        onError: (err) => {
   641→          toast.error(
   642→            err instanceof Error ? err.message : "Failed to invite member",
   643→          );
   644→        },
   645→      },
   646→    );
   647→  };
   648→
   649→  return (
   650→    <form
   651→      onSubmit={handleInvite}
   652→      className="rounded-lg border border-outline-variant bg-surface-container-lowest p-3"
   653→    >
   654→      <div className="mb-2 flex items-center gap-2">
   655→        <Icon name="person_add" size={18} className="text-primary" />
   656→        <span className="text-label-ui-sm uppercase tracking-wider text-on-surface-variant">
   657→          Invite Member
   658→        </span>
   659→      </div>
   660→
   661→      <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">
   662→        <input
   663→          type="email"
   664→          required
   665→          placeholder="Email address"
   666→          value={email}
   667→          onChange={(e) => setEmail(e.target.value)}
   668→          disabled={!canManage || inviteMutation.isPending}
   669→          className="h-9 w-full rounded border border-outline-variant bg-surface px-3 text-body-ui-md text-on-surface transition-colors placeholder:text-on-surface-variant/60 focus:border-primary focus:outline-none disabled:cursor-not-allowed disabled:opacity-50"
   670→        />
   671→        <input
   672→          type="text"
   673→          placeholder="Name (optional)"
   674→          value={name}
   675→          onChange={(e) => setName(e.target.value)}
   676→          maxLength={100}
   677→          disabled={!canManage || inviteMutation.isPending}
   678→          className="h-9 w-full rounded border border-outline-variant bg-surface px-3 text-body-ui-md text-on-surface transition-colors placeholder:text-on-surface-variant/60 focus:border-primary focus:outline-none disabled:cursor-not-allowed disabled:opacity-50"
   679→        />
   680→      </div>
   681→
   682→      <div className="mt-2 flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   683→        <div className="flex items-center gap-2">
   684→          <label
   685→            htmlFor={`invite-role-${orgId}`}
   686→            className="text-label-ui-sm text-on-surface-variant"
   687→          >
   688→            Role
   689→          </label>
   690→          <Select
   691→            value={role}
   692→            onValueChange={(v) => setRole(v as OrgRole)}
   693→            disabled={!canManage || inviteMutation.isPending}
   694→          >
   695→            <SelectTrigger
   696→              id={`invite-role-${orgId}`}
   697→              className="h-9 w-40 border-outline-variant bg-surface text-body-ui-md text-on-surface"
   698→            >
   699→              <SelectValue />
   700→            </SelectTrigger>
   701→            <SelectContent className="bg-surface">
   702→              {INVITE_ROLES.map((r) => {
   703→                const meta = ROLE_META[r];
   704→                return (
   705→                  <SelectItem key={r} value={r}>
   706→                    <span className="flex items-center gap-2">
   707→                      <Icon name={meta.icon} size={14} className={meta.color} />
   708→                      {meta.label}
   709→                    </span>
   710→                  </SelectItem>
   711→                );
   712→              })}
   713→            </SelectContent>
   714→          </Select>
   715→        </div>
   716→
   717→        <button
   718→          type="submit"
   719→          disabled={!canSubmit}
   720→          className="inline-flex h-9 cursor-pointer items-center justify-center gap-2 rounded bg-primary px-4 text-body-ui-md font-medium text-on-primary transition-colors hover:bg-primary-container disabled:cursor-wait disabled:opacity-50"
   721→        >
   722→          <Icon
   723→            name={inviteMutation.isPending ? "progress_activity" : "send"}
   724→            size={16}
   725→            className={inviteMutation.isPending ? "animate-spin" : ""}
   726→          />
   727→          {inviteMutation.isPending ? "Inviting…" : "Send Invite"}
   728→        </button>
   729→      </div>
   730→
   731→      {!canManage && (
   732→        <p className="text-label-ui-sm mt-2 flex items-center gap-1 text-on-surface-variant">
   733→          <Icon name="lock" size={12} />
   734→          Only owners and admins can invite members.
   735→        </p>
   736→      )}
   737→    </form>
   738→  );
   739→}

   741→/* ------------------------------------------------------------------ */
   742→/* Member row                                                          */
   743→/* ------------------------------------------------------------------ */
   744→
   745→function MemberRow({
   746→  member,
   747→  orgId,
   748→  canManage,
   749→}: {
   750→  member: NuswordOrgMember;
   751→  orgId: string;
   752→  canManage: boolean;
   753→}) {
   754→  const updateMutation = useUpdateMember(orgId);
   755→  const removeMutation = useRemoveMember(orgId);
   756→  const [confirmingRemove, setConfirmingRemove] = React.useState(false);
   757→  const roleMeta = ROLE_META[member.role];
   758→  const isOwner = member.role === "owner";
   759→  // Owners can't be removed or have their role changed.
   760→  const immutable = isOwner;
   761→  const canChangeRole = canManage && !immutable;
   762→
   763→  const handleRoleChange = (next: OrgRole) => {
   764→    if (next === member.role) return;
   765→    updateMutation.mutate(
   766→      { memberId: member.id, role: next },
   767→      {
   768→        onSuccess: () => {
   769→          toast.success(`${member.email} is now ${ROLE_META[next].label}`);
   770→        },
   771→        onError: (err) => {
   772→          toast.error(
   773→            err instanceof Error ? err.message : "Failed to update role",
   774→          );
   775→        },
   776→      },
   777→    );
   778→  };
   779→
   780→  const handleRemove = () => {
   781→    removeMutation.mutate(member.id, {
   782→      onSuccess: () => {
   783→        toast.success(`${member.email} removed`);
   784→        setConfirmingRemove(false);
   785→      },
   786→      onError: (err) => {
   787→        toast.error(
   788→          err instanceof Error ? err.message : "Failed to remove member",
   789→        );
   790→      },
   791→    });
   792→  };
   793→
   794→  const displayName = member.name?.trim() || member.email.split("@")[0];
   795→  const initial = displayName.charAt(0).toUpperCase();
   796→
   797→  return (
   798→    <li className="flex items-center gap-3 rounded-lg border border-outline-variant/60 bg-surface px-3 py-2.5 transition-colors hover:border-outline-variant">
   799→      {/* Avatar */}
   800→      <div
   801→        className={cn(
   802→          "flex size-9 shrink-0 items-center justify-center rounded-full text-body-ui-md font-semibold",
   803→          isOwner
   804→            ? "bg-primary text-on-primary"
   805→            : "bg-primary-container text-on-primary-container",
   806→        )}
   807→        aria-hidden="true"
   808→      >
   809→        {initial}
   810→      </div>
   811→
   812→      {/* Name + email */}
   813→      <div className="min-w-0 flex-1">
   814→        <div className="text-body-ui-md truncate font-medium text-on-surface">
   815→          {member.name || displayName}
   816→        </div>
   817→        <div className="text-label-ui-sm truncate text-on-surface-variant">
   818→          {member.email}
   819→        </div>
   820→      </div>
   821→
   822→      {/* Role select / badge */}
   823→      {canChangeRole ? (
   824→        <Select
   825→          value={member.role}
   826→          onValueChange={(v) => handleRoleChange(v as OrgRole)}
   827→          disabled={updateMutation.isPending}
   828→        >
   829→          <SelectTrigger
   830→            className="h-8 w-32 border-outline-variant/60 bg-surface-container-low text-label-ui-sm"
   831→            aria-label={`Change role for ${member.email}`}
   832→          >
   833→            <span className="flex items-center gap-1.5">
   834→              <Icon
   835→                name={roleMeta.icon}
   836→                size={14}
   837→                className={roleMeta.color}
   838→              />
   839→              <SelectValue />
   840→            </span>
   841→          </SelectTrigger>
   842→          <SelectContent className="bg-surface">
   843→            {ALL_ROLES.map((r) => {
   844→              const meta = ROLE_META[r];
   845→              return (
   846→                <SelectItem key={r} value={r}>
   847→                  <span className="flex items-center gap-2">
   848→                    <Icon name={meta.icon} size={14} className={meta.color} />
   849→                    {meta.label}
   850→                  </span>
   851→                </SelectItem>
   852→              );
   853→            })}
   854→          </SelectContent>
   855→        </Select>
   856→      ) : (
   857→        <RoleBadge role={member.role} />
   858→      )}
   859→
   860→      {/* Remove */}
   861→      {canManage && !immutable ? (
   862→        confirmingRemove ? (
   863→          <div className="flex items-center gap-1">
   864→            <button
   865→              type="button"
   866→              onClick={handleRemove}
   867→              disabled={removeMutation.isPending}
   868→              aria-label={`Confirm remove ${member.email}`}
   869→              className="flex size-8 cursor-pointer items-center justify-center rounded bg-destructive text-white transition-colors hover:bg-destructive/90 disabled:cursor-wait disabled:opacity-50"
   870→            >
   871→              <Icon
   872→                name={removeMutation.isPending ? "progress_activity" : "check"}
   873→                size={16}
   874→                className={removeMutation.isPending ? "animate-spin" : ""}
   875→              />
   876→            </button>
   877→            <button
   878→              type="button"
   879→              onClick={() => setConfirmingRemove(false)}
   880→              disabled={removeMutation.isPending}
   881→              aria-label="Cancel remove"
   882→              className="flex size-8 cursor-pointer items-center justify-center rounded text-on-surface-variant transition-colors hover:bg-surface-container-low disabled:opacity-50"
   883→            >
   884→              <Icon name="close" size={16} />
   885→            </button>
   886→          </div>
   887→        ) : (
   888→          <button
   889→            type="button"
   890→            onClick={() => setConfirmingRemove(true)}
   891→            aria-label={`Remove ${member.email}`}
   892→            className="flex size-8 cursor-pointer items-center justify-center rounded text-on-surface-variant transition-colors hover:bg-error-container/40 hover:text-on-error-container"
   893→          >
   894→            <Icon name="person_remove" size={18} />
   895→          </button>
   896→        )
   897→      ) : null}
   898→    </li>
   899→  );
   900→}
