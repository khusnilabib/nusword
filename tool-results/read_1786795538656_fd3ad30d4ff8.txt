    10→ *  - Page settings (Layout panel → paper dimensions/margins, persisted)
    11→ *  - Outline panel (headings extracted from Tiptap doc, click to scroll)
    12→ *  - Basic page count estimation

    14→ *  - Version history (create + restore)

    16→ */

    44→export function EditorView() {
    45→  const documentId = useNuswordStore((s) => s.activeDocumentId);
    46→  const exitToDashboard = useNuswordStore((s) => s.exitToDashboard);
    47→  const mobileLeftOpen = useNuswordStore((s) => s.mobileLeftOpen);
    48→  const mobileRightOpen = useNuswordStore((s) => s.mobileRightOpen);
    49→  const setMobileLeftOpen = useNuswordStore((s) => s.setMobileLeftOpen);
    50→  const setMobileRightOpen = useNuswordStore((s) => s.setMobileRightOpen);
    51→  const anyMobileOpen = mobileLeftOpen || mobileRightOpen;
    52→
    53→  if (!documentId) {
    54→    // Should not happen — dashboard opens editor with an id. Defensive fallback.
    55→    return (
    56→      <div className="flex h-screen items-center justify-center">
    57→        <button
    58→          onClick={exitToDashboard}
    59→          className="rounded border border-outline-variant px-4 py-2 text-body-ui-md text-primary hover:bg-surface-container-low"
    60→        >
    61→          Back to dashboard
    62→        </button>
    63→      </div>
    64→    );
    65→  }
    66→
    67→  return (
    68→    <EditorShell
    69→      documentId={documentId}
    70→      onBack={exitToDashboard}
    71→      mobileLeftOpen={mobileLeftOpen}
    72→      mobileRightOpen={mobileRightOpen}
    73→      setMobileLeftOpen={setMobileLeftOpen}
    74→      setMobileRightOpen={setMobileRightOpen}
    75→      anyMobileOpen={anyMobileOpen}
    76→    />
    77→  );
    78→}

    80→/* ================================================================
    81→   Shell — loads document, manages draft state + autosave
    82→   ================================================================ */
    83→
    84→interface EditorShellProps {
    85→  documentId: string;
    86→  onBack: () => void;
    87→  mobileLeftOpen: boolean;
    88→  mobileRightOpen: boolean;
    89→  setMobileLeftOpen: (v: boolean) => void;
    90→  setMobileRightOpen: (v: boolean) => void;
    91→  anyMobileOpen: boolean;
    92→}
    93→
    94→function EditorShell(props: EditorShellProps) {
    95→  const { documentId, onBack } = props;
    96→  const { data: doc, isLoading, isError } = useDocument(documentId);
    97→
    98→  // Local draft state — the source of truth while editing.
    99→  const [title, setTitle] = React.useState("");
   100→  const [content, setContent] = React.useState<JSONContent | null>(null);
   101→  const [settings, setSettings] = React.useState<PageSettings | null>(null);
   102→  const [hydrated, setHydrated] = React.useState(false);
   103→  const [exportOpen, setExportOpen] = React.useState(false);
   104→  const [shareOpen, setShareOpen] = React.useState(false);
   105→
   106→  // Editor instance — stored in state (not ref) so FindReplace can read it
   107→  // during render without violating the "no ref access during render" rule.
   108→  const [editorInstance, setEditorInstance] =
   109→    React.useState<import("@tiptap/react").Editor | null>(null);
   110→  const editorRef = React.useRef<NuswordEditorHandle | null>(null);
   111→  const onReady = React.useCallback(
   112→    (editor: import("@tiptap/react").Editor | null) => {
   113→      setEditorInstance(editor);
   114→    },
   115→    [],
   116→  );
   117→
   118→  // Hydrate draft from server data (once per document load).
   119→  React.useEffect(() => {
   120→    if (doc) {
   121→      setTitle(doc.title);
   122→      setContent(doc.content);
   123→      setSettings(doc.settings);
   124→      setHydrated(true);
   125→    } else {
   126→      setHydrated(false);
   127→    }
   128→  }, [doc]);
   129→
   130→  const { saveState, flush } = useAutosave({
   131→    documentId,
   132→    title,
   133→    content,
   134→    settings,
   135→    ready: hydrated,
   136→  });
   137→
   138→  // Derive outline + word count from content (recomputed on each change).
   139→  const outline = React.useMemo(() => extractOutline(content), [content]);
   140→  const wordCount = React.useMemo(() => countWordsInDoc(content), [content]);
   141→
   142→  // Pagination — run the deterministic engine on the current content + settings.
   143→  const [measureNonce, setMeasureNonce] = React.useState(0);
   144→  const pagination = usePagination({
   145→    content,
   146→    settings,
   147→    measureNonce,
   148→  });
   149→
   150→  // Re-measure after images load (bump nonce).
   151→  React.useEffect(() => {
   152→    const handler = () => setMeasureNonce((n) => n + 1);
   153→    window.addEventListener("load", handler);
   154→    return () => window.removeEventListener("load", handler);
   155→  }, []);
   156→
   157→  const isRtl = settings?.languageDirection === "rtl";
   158→  const setRtl = (rtl: boolean) => {
   159→    setSettings((s) =>
   160→      s ? { ...s, languageDirection: rtl ? "rtl" : "ltr" } : s,
   161→    );
   162→  };
   163→
   164→  const updateSettings = React.useCallback(
   165→    (patch: Partial<PageSettings>) => {
   166→      setSettings((s) => (s ? { ...s, ...patch } : s));
   167→    },
   168→    [],
   169→  );
   170→
   171→  const editorMode = useNuswordStore((s) => s.editorMode);
   172→  const setEditorMode = useNuswordStore((s) => s.setEditorMode);
   173→  const activePageIndex = useNuswordStore((s) => s.activePageIndex);
   174→  const setActivePageIndex = useNuswordStore((s) => s.setActivePageIndex);
   175→
   176→  if (isLoading) {
   177→    return <EditorLoading />;
   178→  }
   179→  if (isError || !doc || !content || !settings) {
   180→    return (
   181→      <div className="flex h-screen flex-col items-center justify-center gap-3">
   182→        <Icon name="error" size={32} className="text-error" />
   183→        <p className="text-headline-ui-md text-on-surface">
   184→          Couldn&apos;t load document
   185→        </p>
   186→        <button
   187→          onClick={onBack}
   188→          className="rounded border border-outline-variant px-4 py-2 text-body-ui-md text-primary hover:bg-surface-container-low"
   189→        >
   190→          Back to dashboard
   191→        </button>
   192→      </div>
   193→    );
   194→  }
   195→
   196→  return (
   197→    <div className="flex h-screen w-screen flex-col overflow-hidden bg-surface-container-low text-on-surface">
   198→      <EditorTopNav
   199→        title={title}
   200→        onTitleChange={setTitle}
   201→        saveState={saveState}
   202→        onBack={onBack}
   203→        onFlush={flush}
   204→        onOpenMobileLeft={() => props.setMobileLeftOpen(true)}
   205→        onOpenMobileRight={() => props.setMobileRightOpen(true)}
   206→        editorMode={editorMode}
   207→        onToggleMode={() =>
   208→          setEditorMode(editorMode === "edit" ? "preview" : "edit")
   209→        }
   210→        totalPages={pagination.totalPages}
   211→        onExport={() => setExportOpen(true)}
   212→        onShare={() => setShareOpen(true)}
   213→      />
   214→      <div className="relative flex flex-1 overflow-hidden">
   215→        <EditorLeftSidebar
   216→          outline={outline}
   217→          wordCount={wordCount}
   218→          onScrollToHeading={(id) => scrollToHeading(editorInstance, id)}
   219→          content={content}
   220→          settings={settings}
   221→          pagination={pagination}
   222→          activePageIndex={activePageIndex}
   223→          onPageClick={(idx) => {
   224→            setActivePageIndex(idx);
   225→            setEditorMode("preview");
   226→          }}
   227→          onSwitchToPreview={() => setEditorMode("preview")}
   228→        />
   229→        {editorMode === "edit" ? (
   230→          <EditorCanvas
   231→            documentId={documentId}
   232→            content={content}
   233→            onChange={setContent}
   234→            settings={settings}
   235→            title={title}
   236→            onTitleChange={setTitle}
   237→            editorRef={editorRef}
   238→            onReady={onReady}
   239→            editorInstance={editorInstance}
   240→            onMeasureNonce={() => setMeasureNonce((n) => n + 1)}
   241→          />
   242→        ) : (
   243→          <PreviewCanvas
   244→            title={title}
   245→            content={content}
   246→            settings={settings}
   247→            pagination={pagination}
   248→            zoom={useNuswordStore.getState().zoom}
   249→            activePageIndex={activePageIndex}
   250→            onPageClick={setActivePageIndex}
   251→          />
   252→        )}
   253→        <EditorRightSidebar
   254→          settings={settings}
   255→          onUpdateSettings={updateSettings}
   256→          isRtl={isRtl}
   257→          onToggleRtl={() => setRtl(!isRtl)}
   258→        />
   259→
   260→        {/* Mobile backdrop */}
   261→        {props.anyMobileOpen && (
   262→          <div
   263→            role="button"
   264→            tabIndex={0}
   265→            aria-label="Close panels"
   266→            onClick={() => {
   267→              props.setMobileLeftOpen(false);
   268→              props.setMobileRightOpen(false);
   269→            }}
   270→            className="absolute inset-0 z-30 cursor-pointer bg-inverse-surface/40 backdrop-blur-[1px] md:hidden"
   271→          />
   272→        )}
   273→      </div>
   274→      <EditorStatusBar
   275→        wordCount={wordCount}
   276→        isRtl={isRtl}
   277→        onToggleRtl={() => setRtl(!isRtl)}
   278→        totalPages={pagination.totalPages}
   279→      />
   280→      <ExportDialog
   281→        open={exportOpen}
   282→        onOpenChange={setExportOpen}
   283→        documentId={documentId}
   284→        title={title}
   285→        content={content}
   286→        settings={settings}
   287→        pagination={pagination}
   288→      />
   289→      <ShareDialog
   290→        open={shareOpen}
   291→        onOpenChange={setShareOpen}
   292→        documentId={documentId}
   293→        documentTitle={title}
   294→      />
   295→    </div>
   296→  );
   297→}

   299→/* ================================================================
   300→   Top Nav — title input + save state + actions

   302→
   303→function EditorTopNav({
   304→  title,
   305→  onTitleChange,
   306→  saveState,
   307→  onBack,
   308→  onFlush,
   309→  onOpenMobileLeft,
   310→  onOpenMobileRight,
   311→  editorMode,
   312→  onToggleMode,
   313→  totalPages,
   314→  onExport,
   315→  onShare,
   316→}: {
   317→  title: string;
   318→  onTitleChange: (t: string) => void;
   319→  saveState: "idle" | "saving" | "saved" | "error";
   320→  onBack: () => void;
   321→  onFlush: () => void;
   322→  onOpenMobileLeft: () => void;
   323→  onOpenMobileRight: () => void;
   324→  editorMode: "edit" | "preview";
   325→  onToggleMode: () => void;
   326→  totalPages: number;
   327→  onExport: () => void;
   328→  onShare: () => void;
   329→}) {
   330→  const toggleFindReplace = useNuswordStore((s) => s.toggleFindReplace);
   331→
   332→  const saveLabel =
   333→    saveState === "saving"
   334→      ? "Saving…"
   335→      : saveState === "saved"
   336→        ? "Saved"
   337→        : saveState === "error"
   338→          ? "Save failed"
   339→          : "Saved";
   340→  const saveIcon =
   341→    saveState === "saving"
   342→      ? "progress_activity"
   343→      : saveState === "error"
   344→        ? "cloud_off"
   345→        : "cloud_done";
   346→
   347→  return (
   348→    <nav className="z-50 flex h-toolbar-height w-full shrink-0 items-center justify-between border-b border-outline-variant bg-surface px-margin-mobile md:px-gutter">
   349→      {/* Left: back + title + save state */}
   350→      <div className="flex min-w-0 flex-1 items-center gap-2 md:gap-4">
   351→        <button
   352→          type="button"
   353→          onClick={onBack}
   354→          aria-label="Back to dashboard"
   355→          className="flex size-9 shrink-0 cursor-pointer items-center justify-center rounded text-on-surface-variant transition-colors hover:bg-surface-container-low hover:text-primary"
   356→        >
   357→          <Icon name="arrow_back" size={20} />
   358→        </button>
   359→        <button
   360→          type="button"
   361→          onClick={onOpenMobileLeft}
   362→          aria-label="Open navigation"
   363→          className="flex size-9 shrink-0 cursor-pointer items-center justify-center rounded text-on-surface-variant transition-colors hover:bg-surface-container-low hover:text-primary md:hidden"
   364→        >
   365→          <Icon name="menu" size={22} />
   366→        </button>
   367→        <input
   368→          type="text"
   369→          value={title}
   370→          onChange={(e) => onTitleChange(e.target.value)}
   371→          onBlur={onFlush}
   372→          placeholder="Untitled"
   373→          aria-label="Document title"
   374→          className="text-body-ui-md min-w-0 flex-1 truncate border-none bg-transparent text-base italic text-on-surface focus:outline-none md:text-lg"
   375→        />
   376→        <span
   377→          className={cn(
   378→            "text-label-ui-sm hidden items-center gap-1 lg:flex",
   379→            saveState === "error" ? "text-error" : "text-outline",
   380→            saveState === "saving" && "animate-pulse",
   381→          )}
   382→        >
   383→          <Icon name={saveIcon} size={14} />
   384→          {saveLabel}
   385→        </span>
   386→      </div>
   387→
   388→      {/* Right: actions */}
   389→      <div className="flex shrink-0 items-center gap-2">
   390→        {/* Edit / Preview toggle */}
   391→        <div className="flex shrink-0 items-center rounded border border-outline-variant bg-surface-container-low p-0.5">
   392→          <button
   393→            type="button"
   394→            onClick={() => editorMode !== "edit" && onToggleMode()}
   395→            className={cn(
   396→              "flex cursor-pointer items-center gap-1 rounded px-2.5 py-1 text-label-ui-sm transition-colors",
   397→              editorMode === "edit"
   398→                ? "bg-surface-container-highest text-primary"
   399→                : "text-on-surface-variant hover:text-on-surface",
   400→            )}
   401→          >
   402→            <Icon name="edit" size={14} />
   403→            <span className="hidden sm:inline">Edit</span>
   404→          </button>
   405→          <button
   406→            type="button"
   407→            onClick={() => editorMode !== "preview" && onToggleMode()}
   408→            className={cn(
   409→              "flex cursor-pointer items-center gap-1 rounded px-2.5 py-1 text-label-ui-sm transition-colors",
   410→              editorMode === "preview"
   411→                ? "bg-surface-container-highest text-primary"
   412→                : "text-on-surface-variant hover:text-on-surface",
   413→            )}
   414→          >
   415→            <Icon name="visibility" size={14} />
   416→            <span className="hidden sm:inline">Preview</span>
   417→            {totalPages > 0 && (
   418→              <span className="text-mono-ui text-outline">({totalPages})</span>
   419→            )}
   420→          </button>
   421→        </div>
   422→        <button
   423→          type="button"
   424→          onClick={toggleFindReplace}
   425→          aria-label="Find & replace"
   426→          title="Find & replace"
   427→          className="flex size-9 shrink-0 cursor-pointer items-center justify-center rounded text-on-surface-variant transition-colors hover:bg-surface-container-low hover:text-primary"
   428→        >
   429→          <Icon name="search" size={20} />
   430→        </button>
   431→        <button
   432→          type="button"
   433→          onClick={onOpenMobileRight}
   434→          aria-label="Open properties"
   435→          className="flex size-9 shrink-0 cursor-pointer items-center justify-center rounded text-on-surface-variant transition-colors hover:bg-surface-container-low hover:text-primary md:hidden"
   436→        >
   437→          <Icon name="tune" size={20} />
   438→        </button>
   439→        <button
   440→          type="button"
   441→          onClick={onShare}
   442→          className="hidden cursor-pointer rounded border border-outline-variant px-3 py-1.5 text-body-ui-md text-primary transition-colors duration-200 hover:bg-surface-container-low sm:inline-flex"
   443→        >
   444→          Share
   445→        </button>
   446→        <button
   447→          type="button"
   448→          onClick={onExport}
   449→          className="cursor-pointer rounded bg-primary px-3 py-1.5 text-body-ui-md text-on-primary transition-colors duration-200 hover:bg-primary-container sm:px-4"
   450→        >
   451→          Export
   452→        </button>
   453→      </div>
   454→    </nav>
   455→  );
   456→}

   462→function EditorLeftSidebar({
   463→  outline,
   464→  wordCount,
   465→  onScrollToHeading,
   466→  content,
   467→  settings,
   468→  pagination,
   469→  activePageIndex,
   470→  onPageClick,
   471→  onSwitchToPreview,

   474→  wordCount: number;

   506→          <Icon name="close" size={18} />
   507→        </button>
   508→      </div>
   509→
   510→      <div className="flex border-b border-outline-variant">
   511→        <SidebarTab
   512→          active={tab === "outline"}
   513→          onClick={() => setTab("outline")}
   514→          icon="format_list_bulleted"
   515→          label="Outline"
   516→        />
   517→        <SidebarTab
   518→          active={tab === "pages"}
   519→          onClick={() => setTab("pages")}
   520→          icon="layers"
   521→          label="Pages"

   525→          onClick={() => setTab("versions")}
   526→          icon="history"
   527→          label="Versions"
   528→        />
   529→      </div>
   530→
   531→      <div className="flex-1 overflow-y-auto p-4">
   532→        {tab === "outline" ? (
   533→          <OutlinePanel outline={outline} onScrollToHeading={onScrollToHeading} />
   534→        ) : tab === "pages" ? (
   535→          <PageThumbnails
   536→            content={content}
   537→            settings={settings}
   538→            pagination={pagination}
   539→            activePageIndex={activePageIndex}
   540→            onPageClick={onPageClick}
   541→            onSwitchToPreview={onSwitchToPreview}
   542→          />
   543→        ) : (
   544→          <VersionsPanel />
   545→        )}
   546→      </div>
   547→    </aside>
   548→  );

   579→function OutlinePanel({
   580→  outline,
   581→  onScrollToHeading,
   582→}: {
   583→  outline: OutlineEntry[];
   584→  onScrollToHeading: (id: string) => void;
   585→}) {
   586→  if (outline.length === 0) {
   587→    return (
   588→      <div className="flex flex-col items-center justify-center gap-2 py-12 text-center">

   767→    const root = document.querySelector(".nusword-editor-root");
   768→    if (!root) return;
   769→    const handler = () => onMeasureNonce();
   770→    root.addEventListener("load", handler, true); // capture for img load events

   786→          minHeight: mm(heightMm),
   787→          paddingTop: mm(settings.marginTopMm),
   788→          paddingBottom: mm(settings.marginBottomMm),
   789→          paddingLeft: mm(settings.marginLeftMm),
   790→          paddingRight: mm(settings.marginRightMm),
   791→          transform: `scale(${scale})`,
   792→          columnCount: settings.columns > 1 ? settings.columns : undefined,

  1489→/** Scroll the Tiptap editor to a heading by its slug id. */
  1490→function scrollToHeading(
  1491→  editor: import("@tiptap/react").Editor | null,
  1492→  id: string,
  1493→) {
  1494→  if (!editor) return;
  1495→  const { state } = editor;
  1496→  let foundPos: number | null = null;
  1497→  state.doc.descendants((node, pos) => {
  1498→    if (node.type.name !== "heading") return;
  1499→    const text = node.textContent;
  1500→    const slug =
  1501→      "h-" +
  1502→      text
  1503→        .toLowerCase()
  1504→        .replace(/[^\w\s-]/g, "")
  1505→        .trim()
  1506→        .replace(/\s+/g, "-")
  1507→        .slice(0, 40);
  1508→    if (slug === id) {
  1509→      foundPos = pos;
  1510→      return false;
  1511→    }
  1512→  });
  1513→  if (foundPos !== null) {
  1514→    editor.commands.setTextSelection(foundPos);
  1515→    editor.commands.scrollIntoView();
  1516→  }
  1517→}
